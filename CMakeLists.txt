# Switch build options. tools/build-switch.sh sets them from its SWITCH_* variables (docs/BUILDING-SWITCH.md).
# The default build is the one before these options existed, apart from the recompiled-code flags in
# MarathonRecompLib/CMakeLists.txt and -ffixed-x16. Note that the libraries under tools/ (fmt, zstd,
# XenonUtils, disasm) are already compiled as LTO objects (tools/CMakeLists.txt turns IPO on), so every
# link of the app runs lto-wrapper for them, with or without MARATHON_RECOMP_SWITCH_LTO.
if (MARATHON_RECOMP_SWITCH)
    # Link-time optimisation of the whole app. Its benefit comes with tools/switch-direct-calls.py, which lets
    # GCC inline across recompiled functions at all. Balanced partitions and 2 LTRANS jobs keep the link of the
    # recompiled game within the RAM of a normal build PC.
    option(MARATHON_RECOMP_SWITCH_LTO "Build the Switch NRO with link-time optimisation." OFF)
    set(MARATHON_RECOMP_SWITCH_LTO_JOBS "2" CACHE STRING "Parallel LTRANS jobs for the Switch LTO link.")

    # -O2 instead of CMake's release -O3: for all the code, or for the recompiled code only. The Cortex-A57 has
    # a 48 KB instruction cache and -O3 grows the recompiled code; with PGO, GCC turns the -O3 work back on
    # for the code the profile finds hot.
    option(MARATHON_RECOMP_SWITCH_O2 "Compile all Switch code with -O2 instead of -O3." OFF)
    option(MARATHON_RECOMP_SWITCH_RECOMP_O2 "Compile the recompiled game code with -O2 instead of -O3." OFF)

    # Interprocedural points-to analysis; it runs on the whole program at the LTO link, so it goes to the link too.
    option(MARATHON_RECOMP_SWITCH_IPA_PTA "Compile and link the Switch build with -fipa-pta." OFF)

    # ppc/ generated by XenonRecomp in its classic mode (XENON_RECOMP_CLASSIC=1), whose code expects guest
    # memory accesses to be volatile (ppc_context.h); the app reads the same macros. For A/B measurements.
    option(MARATHON_RECOMP_SWITCH_CLASSIC_CODEGEN "ppc/ was generated with the classic code generation." OFF)

    # x16 kept out of register allocation in the recompiled code and the app: os/switch/exception_switch.cpp
    # resumes an emulated guest access through x16 (ResumeGuestContext), so no value may live in it there.
    option(MARATHON_RECOMP_SWITCH_FIXED_X16 "Reserve x16 in the Switch app and recompiled code (-ffixed-x16)." ON)

    # Opt-in: Xbox 360-exact fused multiply-add rounding. It changes the last bit of some float results compared
    # with builds without it. The guest's fused instructions (fmadd, vmaddfp...), which XenonRecomp writes as
    # explicit fma (PPC_FMA, simde_mm_vmaddfp), are fused (PPC_EXPLICIT_FMA), and the recompiled code is built
    # with -ffp-contract=off, so no other multiply and add are fused; one rounding per guest instruction, as on
    # the Xbox 360. Off, GCC's default contraction applies as in the base port: it also fuses some separate guest
    # multiplies and adds, and which ones depends on the surrounding code and on inlining.
    # Only used when the generated ppc/ text has the explicit fma (checked below): not with the classic code
    # generation, nor with a XenonRecomp without patches/XenonRecomp-switch-perf.patch, where -ffp-contract=off
    # would round every guest fused operation twice.
    # Not exact with MARATHON_RECOMP_SWITCH_LTO: a guest function that LTO inlines into an app hook (which calls
    # its __imp__sub_X directly) is compiled with the app's default contraction there, which can fuse its
    # separate multiplies and adds again (docs/BUILDING-SWITCH.md).
    option(MARATHON_RECOMP_SWITCH_EXACT_FMA "Xbox 360-exact fused multiply-add rounding; changes the last bit of some float results compared with builds without it." OFF)

    # Plain (non-volatile) guest memory accesses in the recompiled code (MarathonRecompLib/CMakeLists.txt).
    option(MARATHON_RECOMP_SWITCH_PLAIN_GUEST_MEMORY "Recompiled code with plain guest memory accesses and loop barriers." OFF)
    set(MARATHON_RECOMP_SWITCH_CODEGEN_OPTIONS "" CACHE STRING "The code generation options ppc/ was written with (tools/build-switch.sh), for the build options line.")

    # Profile-guided optimisation (applied in MarathonRecomp/CMakeLists.txt): empty, generate or use.
    set(MARATHON_RECOMP_SWITCH_PGO "" CACHE STRING "Switch PGO: empty, generate or use")
    set(MARATHON_RECOMP_SWITCH_PGO_DIR "${CMAKE_SOURCE_DIR}/pgo" CACHE PATH "Folder holding the .gcda profile")

    # Written into the generated switch_build_id.h (MarathonRecomp/CMakeLists.txt), so a log names its build.
    set(MARATHON_RECOMP_SWITCH_BUILD_ID "" CACHE STRING "Switch build identifier (tools/build-switch.sh sets it).")

    if (MARATHON_RECOMP_SWITCH_LTO)
        message(STATUS "Switch: LTO (${MARATHON_RECOMP_SWITCH_LTO_JOBS} jobs)")
        add_compile_options(-flto -fno-fat-lto-objects)
        add_link_options(-flto=${MARATHON_RECOMP_SWITCH_LTO_JOBS} -flto-partition=balanced)
        add_compile_definitions(MARATHON_RECOMP_SWITCH_LTO)
    endif()

    if (MARATHON_RECOMP_SWITCH_O2)
        message(STATUS "Switch: -O2")
        string(REPLACE "-O3" "-O2" CMAKE_C_FLAGS_RELEASE "${CMAKE_C_FLAGS_RELEASE}")
        string(REPLACE "-O3" "-O2" CMAKE_CXX_FLAGS_RELEASE "${CMAKE_CXX_FLAGS_RELEASE}")
        string(REPLACE "-O3" "-O2" CMAKE_C_FLAGS_RELWITHDEBINFO "${CMAKE_C_FLAGS_RELWITHDEBINFO}")
        string(REPLACE "-O3" "-O2" CMAKE_CXX_FLAGS_RELWITHDEBINFO "${CMAKE_CXX_FLAGS_RELWITHDEBINFO}")
    endif()

    if (MARATHON_RECOMP_SWITCH_IPA_PTA)
        message(STATUS "Switch: -fipa-pta")
        add_compile_options(-fipa-pta)
        add_link_options(-fipa-pta)
    endif()

    # Whether exact FMA is really used: MarathonRecompLib/CMakeLists.txt (-ffp-contract=off) and the build options
    # string (MarathonRecomp/CMakeLists.txt) read this, not the option.
    set(MARATHON_RECOMP_SWITCH_EXACT_FMA_ACTIVE OFF)
    if (MARATHON_RECOMP_SWITCH_EXACT_FMA)
        set(MARATHON_RECOMP_PPC_DIR "${CMAKE_SOURCE_DIR}/MarathonRecompLib/ppc")
        # The same choice MarathonRecompLib/CMakeLists.txt makes for its option of that name.
        if (DEFINED MARATHON_RECOMP_SWITCH_USE_PREGENERATED_PPC)
            set(MARATHON_RECOMP_PPC_PREGENERATED ${MARATHON_RECOMP_SWITCH_USE_PREGENERATED_PPC})
        elseif (EXISTS "${MARATHON_RECOMP_PPC_DIR}/ppc_config.h")
            set(MARATHON_RECOMP_PPC_PREGENERATED ON)
        else()
            set(MARATHON_RECOMP_PPC_PREGENERATED OFF)
        endif()
        if (MARATHON_RECOMP_SWITCH_CLASSIC_CODEGEN)
            message(WARNING "MARATHON_RECOMP_SWITCH_EXACT_FMA is ignored with MARATHON_RECOMP_SWITCH_CLASSIC_CODEGEN "
                "(the classic text has no explicit fma): GCC's default contraction is used.")
        elseif (NOT MARATHON_RECOMP_PPC_PREGENERATED)
            message(WARNING "MARATHON_RECOMP_SWITCH_EXACT_FMA needs ppc/ generated before the configure step "
                "(tools/build-switch.sh does that), so that its text can be checked: GCC's default contraction is used.")
        else()
            # The first file with an explicit fma, normally ppc_recomp.0.cpp; text without any is scanned whole.
            set(MARATHON_RECOMP_PPC_FMA_FILE "")
            set(MARATHON_RECOMP_PPC_SCANNED "${MARATHON_RECOMP_PPC_DIR}/ppc_context.h")
            set(ppc_file_index 0)
            while (EXISTS "${MARATHON_RECOMP_PPC_DIR}/ppc_recomp.${ppc_file_index}.cpp")
                set(ppc_file "${MARATHON_RECOMP_PPC_DIR}/ppc_recomp.${ppc_file_index}.cpp")
                list(APPEND MARATHON_RECOMP_PPC_SCANNED "${ppc_file}")
                file(STRINGS "${ppc_file}" ppc_fma_line REGEX "PPC_FM[AS][(]|simde_mm_v(madd|nmsub)fp[(]" LIMIT_COUNT 1)
                if (ppc_fma_line)
                    set(MARATHON_RECOMP_PPC_FMA_FILE "${ppc_file}")
                    break()
                endif()
                math(EXPR ppc_file_index "${ppc_file_index} + 1")
            endwhile()
            # A regenerated ppc/ (other mode or recompiler) is checked again.
            set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${MARATHON_RECOMP_PPC_SCANNED})
            if (MARATHON_RECOMP_PPC_FMA_FILE)
                set(MARATHON_RECOMP_SWITCH_EXACT_FMA_ACTIVE ON)
            else()
                message(WARNING "MARATHON_RECOMP_SWITCH_EXACT_FMA is ignored: ${MARATHON_RECOMP_PPC_DIR} has no explicit "
                    "fma (generated without patches/XenonRecomp-switch-perf.patch, or in classic mode), and "
                    "-ffp-contract=off would round every guest fused operation twice. GCC's default contraction is used.")
            endif()
        endif()
    endif()

    if (MARATHON_RECOMP_SWITCH_CLASSIC_CODEGEN)
        message(STATUS "Switch: classic guest code generation")
        add_compile_definitions(PPC_VOLATILE_GUEST_MEMORY MARATHON_RECOMP_CLASSIC_CODEGEN)
    elseif (MARATHON_RECOMP_SWITCH_EXACT_FMA_ACTIVE)
        # Defined for every target: the fma helpers are inline functions in ppc_context.h, which the app includes too.
        message(STATUS "Switch: exact guest fused multiply-adds")
        add_compile_definitions(PPC_EXPLICIT_FMA)
    endif()
endif()
